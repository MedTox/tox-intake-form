/** * Toxicology Teleconsult Intake — Google Sheet webhook (with photo support) * * This replaces the existing Code.gs for your main teleconsult * Google Sheet. It does NOT change which sheet you're connected to — * it only adds the ability to save uploaded photos to Drive and link * them in new columns. New fields (Region, photo links, etc.) will * appear as new columns automatically, same as before. */ const SHEET_NAME = 'Submissions'; const PHOTOS_FOLDER_NAME = 'Toxicology Teleconsult Photos'; // Shared secret — must match SHARED_SECRET in the HTML form. Unchanged. const SHARED_SECRET = 'dd1f1aef222c7deb8199d32fc2fa4f2f'; function getOrCreatePhotosFolder() { const folders = DriveApp.getFoldersByName(PHOTOS_FOLDER_NAME); return folders.hasNext() ? folders.next() : DriveApp.createFolder(PHOTOS_FOLDER_NAME); } function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (data['_secret'] !== SHARED_SECRET) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Unauthorized' })) .setMimeType(ContentService.MimeType.JSON); } delete data['_secret']; // Save any uploaded photos to Drive and replace them with view links, // grouped by category (Medication / Site / ECG / Imaging) into separate columns const images = data['_images']; delete data['_images']; if (images && images.length) { const folder = getOrCreatePhotosFolder(); const serial = data['Serial No.'] || 'unknown'; const categoryCounters = {}; images.forEach((img) => { const category = img.category || 'Photo'; categoryCounters[category] = (categoryCounters[category] || 0) + 1; const idx = categoryCounters[category]; const bytes = Utilities.base64Decode(img.data); const blob = Utilities.newBlob(bytes, img.mimeType || 'image/jpeg', serial + '_' + category + '_' + idx + '.jpg'); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); data['Photo (' + category + ') ' + idx] = file.getUrl(); }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); data['Received At'] = new Date(); // First submission ever: write the header row from the incoming keys if (sheet.getLastRow() === 0) { sheet.appendRow(Object.keys(data)); } // Align each row to the existing header order, adding any new // columns the form might send in the future to the end. let header = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const newKeys = Object.keys(data).filter(k => header.indexOf(k) === -1); if (newKeys.length) { sheet.getRange(1, header.length + 1, 1, newKeys.length).setValues([newKeys]); header = header.concat(newKeys); } const row = header.map(h => (h in data ? data[h] : '')); sheet.appendRow(row); return ContentService .createTextOutput(JSON.stringify({ status: 'ok' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } // Optional: lets you sanity-check the deployment by visiting the // web app URL directly in a browser (GET request). function doGet(e) { return ContentService.createTextOutput('Toxicology intake webhook is live.'); }
